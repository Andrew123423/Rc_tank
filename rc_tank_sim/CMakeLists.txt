cmake_minimum_required(VERSION 3.16)
project(rc_tank_sim)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_msgs_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_library(tank_dynamics SHARED src/tank_dynamics.cpp)
target_link_libraries(tank_dynamics gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
add_executable(firmware_controller src/firmware_controller.cpp)
target_include_directories(firmware_controller PRIVATE include vendor)
ament_target_dependencies(firmware_controller rclcpp std_msgs geometry_msgs)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS tank_dynamics DESTINATION lib)
install(TARGETS firmware_controller DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/keyboard_teleop scripts/state_visualizer scripts/smoke_check scripts/diagnose_runtime DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config meshes worlds rviz docs urdf DESTINATION share/${PROJECT_NAME})
install(FILES README_RU.md THIRD_PARTY.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(firmware_test test/firmware_test.cpp)
  target_include_directories(firmware_test PRIVATE include vendor)
  add_test(NAME firmware_logic COMMAND firmware_test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME launch_wiring COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_launch_config.py)
  add_test(NAME teleop_and_geometry COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_controls.py)
  add_test(NAME model_structure COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_model.py)
endif()
ament_package()
